    <script>
        /* ==========================================
           1. SPATIAL TOPOLOGY & ROUTING DEFINITION
           ========================================== */
        // Explicit coordinates representing 1:1 indoor scale coordinates in 3D space.
        const nodes = {
            // Floor 0 (Elevations Y = 0)
            'f0_lobby': { x: 0, y: 0, z: 12, floor: 0, label: 'Floor 0 - Main Lobby' },
            'f0_office_west': { x: -16, y: 0, z: 5, floor: 0, label: 'Floor 0 - Workspace West' },

            'f0_corridor': { x: 0, y: 0, z: 0, floor: 0, label: 'Floor 0 - Hallway Connector' },
            
            // Floor 1 (Elevations Y = 10)
            'f1_stairs': { x: 8, y: 10, z: 0, floor: 1, label: 'Floor 1 - Stairwell Landing' },
            'f1_elevator': { x: -12, y: 10, z: -12, floor: 1, label: 'Floor 1 - Elevator Portal' },
            'f1_exec_suite': { x: 12, y: 10, z: 8, floor: 1, label: 'Floor 1 - Executive Suite 101' },
            'f1_corridor': { x: 0, y: 10, z: 0, floor: 1, label: 'Floor 1 - Hallway Connector' },
            'f1_stairs_up': { x: 8, y: 10, z: -4, floor: 1, label: 'Floor 1 - Stairwell (To F2)' },
            
            // Floor 2 (Elevations Y = 20)
            'f2_stairs': { x: 8, y: 20, z: -4, floor: 2, label: 'Floor 2 - Stairwell Landing' },
            'f2_elevator': { x: -12, y: 20, z: -12, floor: 2, label: 'Floor 2 - Elevator Hub' },
            'f2_boardroom': { x: 6, y: 20, z: 12, floor: 2, label: 'Floor 2 - Executive Boardroom' },
            'f2_cafeteria': { x: -14, y: 20, z: 8, floor: 2, label: 'Floor 2 - Sky Cafeteria' },
            'f2_corridor': { x: 0, y: 20, z: 0, floor: 2, label: 'Floor 2 - Hallway Connector' }
        };

        // Graph connections (bidirectional) with vertical type markers to allow accessibility filtering
        const connections = [
            // Floor 0 local
            { from: 'f0_lobby', to: 'f0_corridor', type: 'flat' },
            { from: 'f0_office_west', to: 'f0_corridor', type: 'flat' },
            { from: 'f0_elevator', to: 'f0_corridor', type: 'flat' },
            { from: 'f0_stairs', to: 'f0_corridor', type: 'flat' },

            // Floor 0 to Floor 1 Stairwell Connector
            { from: 'f0_stairs', to: 'f1_stairs', type: 'stairs' },

            // Floor 1 local
            { from: 'f1_stairs', to: 'f1_corridor', type: 'flat' },
            { from: 'f1_elevator', to: 'f1_corridor', type: 'flat' },
            { from: 'f1_exec_suite', to: 'f1_corridor', type: 'flat' },
            { from: 'f1_stairs_up', to: 'f1_corridor', type: 'flat' },

            // Floor 1 to Floor 2 Stairwell Connector
            { from: 'f1_stairs_up', to: 'f2_stairs', type: 'stairs' },

            // Floor 2 local
            { from: 'f2_stairs', to: 'f2_corridor', type: 'flat' },
            { from: 'f2_elevator', to: 'f2_corridor', type: 'flat' },
            { from: 'f2_boardroom', to: 'f2_corridor', type: 'flat' },
            { from: 'f2_cafeteria', to: 'f2_corridor', type: 'flat' },

            // Elevator Portals (Vertical connection bridges)
            { from: 'f0_elevator', to: 'f1_elevator', type: 'elevator' },
            { from: 'f1_elevator', to: 'f2_elevator', type: 'elevator' }
        ];

        // Core App State Configuration
        let routingProfile = 'standard'; // 'standard' or 'accessibility'
        let startNodeId = 'f0_lobby';
        let destNodeId = '';
        let floorVisibility = { 0: true, 1: true, 2: true };
        
        // 3D Runtime Global Variables
        let scene, camera, renderer, controls;
        let floorPlates = {};
        let architecturalWalls = [];
        let waypointVisuals = {};
        let activePathMesh = null;
        let elevatorCabMesh = null;
        
        // Simulation details
        let simulationActive = false;
        let simInterval = null;
        let simMarker = null;
        let currentSimPath = [];
        let currentSimIndex = 0;

        /* ==========================================
           2. DIJKSTRA / A* ENGINE FOR WEB PATHS
           ========================================== */
        function solvePathfinding(start, target, profile) {
            // Build adjacency list dynamically depending on active profile constraints
            const adjacency = {};
            for (const nodeId in nodes) {
                adjacency[nodeId] = [];
            }

            connections.forEach(edge => {
                // If Profile = Accessibility (ADA), completely ignore stairway connections
                if (profile === 'accessibility' && edge.type === 'stairs') {
                    return; 
                }

                const nA = nodes[edge.from];
                const nB = nodes[edge.to];
                
                // Real distance calculation (Euclidean metrics)
                let weight = Math.sqrt(
                    Math.pow(nA.x - nB.x, 2) +
                    Math.pow(nA.y - nB.y, 2) +
                    Math.pow(nA.z - nB.z, 2)
                );

                // Add wait penalties on portals to emulate real-world wait times
                if (edge.type === 'elevator') {
                    weight += 8.0; // Simulated elevator vertical latency delay
                } else if (edge.type === 'stairs') {
                    weight *= 1.3; // Scaling penalty for stair elevation climbing
                }

                adjacency[edge.from].push({ node: edge.to, weight });
                adjacency[edge.to].push({ node: edge.from, weight });
            });

            // Standard Dijkstra algorithm
            const distances = {};
            const previous = {};
            const unvisited = new Set();

            for (const nodeId in nodes) {
                distances[nodeId] = Infinity;
                previous[nodeId] = null;
                unvisited.add(nodeId);
            }
            distances[start] = 0;

            while (unvisited.size > 0) {
                // Find node with smallest distance
                let minNode = null;
                unvisited.forEach(nodeId => {
                    if (minNode === null || distances[nodeId] < distances[minNode]) {
                        minNode = nodeId;
                    }
                });

                if (minNode === null || distances[minNode] === Infinity) break;
                if (minNode === target) break;

                unvisited.delete(minNode);

                adjacency[minNode].forEach(neighbor => {
                    if (unvisited.has(neighbor.node)) {
                        const newDist = distances[minNode] + neighbor.weight;
                        if (newDist < distances[neighbor.node]) {
                            distances[neighbor.node] = newDist;
                            previous[neighbor.node] = minNode;
                        }
                    }
                });
            }

            // Reconstruction
            const path = [];
            let current = target;
            while (current !== null) {
                path.unshift(current);
                current = previous[current];
            }

            return path[0] === start ? path : [];
        }

        // Generate high-fidelity turn-by-turn navigation instructions
        function generateRouteDirections(path) {
            if (path.length === 0) return { main: 'Target Unreachable', sub: 'Adjust routing profiling options.' };
            
            let steps = [];
            let stairsCount = 0;
            let elevatorCount = 0;
            
            for (let i = 0; i < path.length - 1; i++) {
                const nodeA = nodes[path[i]];
                const nodeB = nodes[path[i+1]];
                
                // Identify connection characteristics
                const connectionEdge = connections.find(c => 
                    (c.from === path[i] && c.to === path[i+1]) || 
                    (c.from === path[i+1] && c.to === path[i])
                );
                
                if (connectionEdge && connectionEdge.type === 'stairs') {
                    stairsCount++;
                } else if (connectionEdge && connectionEdge.type === 'elevator') {
                    elevatorCount++;
                }
            }

            const finalTarget = nodes[path[path.length - 1]];
            
            let mainStr = `Proceed to ${finalTarget.label}`;
            let subStr = '';

            if (elevatorCount > 0 && stairsCount > 0) {
                subStr = `Dynamic multi-level route using elevator & stairs.`;
            } else if (elevatorCount > 0) {
                subStr = `Direct elevator access route (Fully ADA Compliant).`;
            } else if (stairsCount > 0) {
                subStr = `Staircase climbing required across levels.`;
            } else {
                subStr = `Level navigation across same floor layout.`;
            }

            return { main: mainStr, sub: subStr };
        }

        function init3DEngine() {
            const container = document.getElementById('three-container');
            const width = container.clientWidth;
            const height = container.clientHeight;

            // Scene and Sky environment
            scene = new THREE.Scene();
            scene.background = new THREE.Color('#060913');
            scene.fog = new THREE.FogExp2('#060913', 0.005);

            // Camera setup - High precision Orthographic or isometric look
            camera = new THREE.PerspectiveCamera(45, width / height, 1, 1000);
            camera.position.set(40, 50, 60);

            // Renderer setup
            renderer = new THREE.WebGLRenderer({ antialias: true, alpha: false });
            renderer.setSize(width, height);
            renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
            renderer.shadowMap.enabled = true;
            renderer.shadowMap.type = THREE.PCFSoftShadowMap;
            container.appendChild(renderer.domElement);

            // Controls setup
            controls = new THREE.OrbitControls(camera, renderer.domElement);
            controls.enableDamping = true;
            controls.dampingFactor = 0.05;
            controls.maxPolarAngle = Math.PI / 2 - 0.05; // Prevent camera from dipping below structural floor plane
            controls.minDistance = 15;
            controls.maxDistance = 180;

            // Ambient, direction, & atmospheric key lighting
            const ambientLight = new THREE.AmbientLight('#1e293b', 0.8);
            scene.add(ambientLight);

            const directLight = new THREE.DirectionalLight('#06b6d4', 1.8);
            directLight.position.set(30, 80, 40);
            directLight.castShadow = true;
            directLight.shadow.mapSize.width = 2048;
            directLight.shadow.mapSize.height = 2048;
            directLight.shadow.bias = -0.0001;
            scene.add(directLight);

            const secondaryLight = new THREE.PointLight('#f59e0b', 0.8, 100);
            secondaryLight.position.set(-30, 20, -30);
            scene.add(secondaryLight);

            // Build architecture mockup
            generateBuildingMockup();

            // Setup waypoint indicators
            generateWaypoints();

            // Window Resizing handlers
            window.addEventListener('resize', onWindowResize);

            // Initialize rendering loop
            animate();
        }

        function onWindowResize() {

            const height = container.clientHeight;
            
            camera.aspect = width / height;

            renderer.setSize(width, height);
        }

        function generateBuildingMockup() {
            const floorMaterial = new THREE.MeshStandardMaterial({
                color: '#111827',
                roughness: 0.7,
                metalness: 0.1,
                transparent: true,
                opacity: 0.75
            });

            const floorBorderMaterial = new THREE.LineBasicMaterial({ color: '#1e293b', linewidth: 2 });
            const wallsMaterial = new THREE.MeshStandardMaterial({
                color: '#0f172a',
                roughness: 0.8,
                metalness: 0.1,
                transparent: true,
                opacity: 0.35,
                wireframe: false
            });

            const floorPlatesCount = 3;
            const heightMultiplier = 10; // Vertical gap between floor levels in Three.js coordinates

            for (let f = 0; f < floorPlatesCount; f++) {
                // Group each floor's interior models for modular visibility
                const floorGroup = new THREE.Group();
                
                // Floor plane (Base)
                const geometry = new THREE.BoxGeometry(45, 0.4, 35);
                const baseMesh = new THREE.Mesh(geometry, floorMaterial);
                baseMesh.position.y = f * heightMultiplier;
                baseMesh.receiveShadow = true;
                floorGroup.add(baseMesh);

                // Tech-styled outer grid framing
                const edges = new THREE.EdgesGeometry(geometry);
                const line = new THREE.LineSegments(edges, floorBorderMaterial);
                line.position.y = f * heightMultiplier;
                floorGroup.add(line);

                // Add room divider mock walls
                generateRoomWalls(floorGroup, f, wallsMaterial);

                scene.add(floorGroup);
                floorPlates[f] = floorGroup;
            }

            // Create structural Elevator Shaft model
            const shaftGeometry = new THREE.BoxGeometry(4, 30, 4);
            const shaftEdges = new THREE.EdgesGeometry(shaftGeometry);
            const shaftBorderMat = new THREE.LineBasicMaterial({ color: '#06b6d4', transparent: true, opacity: 0.4 });
            const shaftLine = new THREE.LineSegments(shaftEdges, shaftBorderMat);
            shaftLine.position.set(-12, 10, -12);
            scene.add(shaftLine);

            // Create active wireframe elevator cab mesh
            const cabGeometry = new THREE.BoxGeometry(3.2, 3.2, 3.2);
            const cabMaterial = new THREE.MeshStandardMaterial({
                color: '#06b6d4',
                roughness: 0.2,
                metalness: 0.8,
                transparent: true,
                opacity: 0.6
            });
            elevatorCabMesh = new THREE.Mesh(cabGeometry, cabMaterial);
            elevatorCabMesh.position.set(-12, 0, -12);
            scene.add(elevatorCabMesh);

            // Create staircase geometry visuals (spiral path wrapper representation)
            const staircaseMat = new THREE.MeshStandardMaterial({ color: '#f59e0b', roughness: 0.5 });
            for (let f = 0; f < floorPlatesCount - 1; f++) {
                const yOffset = f * heightMultiplier;
                const steps = 14;
                for (let i = 0; i < steps; i++) {
                    const progress = i / steps;
                    const stepGeom = new THREE.BoxGeometry(3, 0.2, 1);
                    const stepMesh = new THREE.Mesh(stepGeom, staircaseMat);
                    
                    // Arc rotation
                    const angle = progress * Math.PI * 0.8;
                    const radius = 3.5;
                    stepMesh.position.set(8 + Math.cos(angle) * radius, yOffset + progress * heightMultiplier, Math.sin(angle) * radius);
                    stepMesh.rotation.y = -angle;
                    scene.add(stepMesh);
                }
            }
        }

        function generateRoomWalls(group, floorIndex, material) {
            // Helper to generate simple wall bounding blockages
            const wallCreator = (w, h, d, x, z, angle = 0) => {
                const geometry = new THREE.BoxGeometry(w, h, d);
                const wall = new THREE.Mesh(geometry, material);
                wall.position.set(x, (floorIndex * 10) + h/2, z);
                wall.rotation.y = angle;
                wall.userData = { floor: floorIndex };
                group.add(wall);
                architecturalWalls.push(wall);
            };

            // Setup walls depending on logical layout of different floor plates
            if (floorIndex === 0) {
                wallCreator(0.4, 6, 12, -8, 8); // West office divider
                wallCreator(12, 6, 0.4, -14, 2); // West office inner
                wallCreator(0.4, 6, 8, 4, 10, Math.PI / 4); // Lobby corridor diagonal
            } else if (floorIndex === 1) {
                wallCreator(15, 6, 0.4, 4, 4); // Corridor partition
                wallCreator(0.4, 6, 10, -3, -1); 
            } else if (floorIndex === 2) {
                wallCreator(0.4, 6, 15, -4, 4); // Cafeteria glass partition
                wallCreator(12, 6, 0.4, 4, -4); // Boardroom rear wall
            }
        }

        function generateWaypoints() {
            // Base physical pins indicating navigational target junctions
            const pinGeometry = new THREE.CylinderGeometry(0.0, 0.4, 1.2, 6);
            const pinMaterial = new THREE.MeshStandardMaterial({
                color: '#334155',
                metalness: 0.8,
                roughness: 0.2
            });

            for (const key in nodes) {
                const node = nodes[key];
                
                // Exclude raw transition nodes like hallways/corridors from drawing big target indicators
                if (key.includes('corridor') || key.includes('stairs_up')) continue;

                const pinMesh = new THREE.Mesh(pinGeometry, pinMaterial);
                // Invert pin so point faces directly towards waypoint floor plane
                pinMesh.rotation.x = Math.PI;
                pinMesh.position.set(node.x, node.y + 1.5, node.z);
                
                scene.add(pinMesh);
                waypointVisuals[key] = pinMesh;
            }
        }

        function calculateRoute() {
            // Collect selected form options
            startNodeId = document.getElementById('start-select').value;
            destNodeId = document.getElementById('destination-select').value;

            if (!destNodeId) {
                clearRouting();
                return;
            }

            document.getElementById('telemetry-node').innerText = startNodeId;

            // Run Dijkstra solver logic
            const rawPath = solvePathfinding(startNodeId, destNodeId, routingProfile);

            if (rawPath.length === 0) {
                document.getElementById('telemetry-status').className = 'text-brand-amber font-semibold';
                document.getElementById('telemetry-status').innerText = 'Unreachable Destination';
                clearRoutingMesh();
                return;
            }

            // Route directions text update
            const directions = generateRouteDirections(rawPath);
            document.getElementById('nav-step-text').innerText = directions.main;
            document.getElementById('nav-step-subtext').innerText = directions.sub;

            // Animate dynamic overlay panels in
            const navOverlay = document.getElementById('nav-overlay');
            navOverlay.style.opacity = '1';
            navOverlay.style.pointerEvents = 'auto';

            // Measure accurate cumulative distance metrics
            let totalDist = 0;
            for (let i = 0; i < rawPath.length - 1; i++) {
                const nA = nodes[rawPath[i]];
                const nB = nodes[rawPath[i+1]];
                totalDist += Math.sqrt(
                    Math.pow(nA.x - nB.x, 2) +
                    Math.pow(nA.y - nB.y, 2) +
                    Math.pow(nA.z - nB.z, 2)
                );
            }

            document.getElementById('telemetry-status').className = 'text-brand-emerald font-semibold';
            document.getElementById('telemetry-status').innerText = 'Optimal Routing Active';
            document.getElementById('telemetry-distance').innerText = `${totalDist.toFixed(1)}m`;
            
            // Average structural pedestrian walking velocity is 1.4 m/s
            const walkTimeSeconds = Math.round(totalDist / 1.2);
            document.getElementById('telemetry-time').innerText = `${walkTimeSeconds}s`;

            // Setup path spline rendering in 3D scene
            drawPathLine(rawPath);

            // Populate simulation parameters
            currentSimPath = rawPath;
            currentSimIndex = 0;
            document.getElementById('sim-node-counter').innerText = `0/${currentSimPath.length}`;

            // Highlight target waypoint visual pins
            resetAllWaypointPinColors();
            setWaypointPinColor(startNodeId, '#10b981'); // Emerald Start
            setWaypointPinColor(destNodeId, '#06b6d4'); // Cyan Target
        }

        // Catmull-Rom math to smooth out jagged corner vectors on pathlines
        function drawPathLine(pathNodeIds) {
            clearRoutingMesh();

            const pathVectors = [];
            pathNodeIds.forEach(nodeId => {
                const n = nodes[nodeId];
                // Lift path slightly off floor plates to prevent visual clipping (z-fighting)
                pathVectors.push(new THREE.Vector3(n.x, n.y + 0.5, n.z));
            });

            // If only 2 nodes, linear line, else generate cubic spline curve
            let curvePoints = [];
            if (pathVectors.length < 3) {
                curvePoints = pathVectors;
            } else {
                const curve = new THREE.CatmullRomCurve3(pathVectors);
                curve.curveType = 'centripetal';
                curvePoints = curve.getPoints(100); // Tessellate curve points
            }

            const geometry = new THREE.BufferGeometry().setFromPoints(curvePoints);
            
            // Dynamic neon glow animating material
            const pathMaterial = new THREE.LineDashedMaterial({
                color: '#06b6d4',
                linewidth: 5,
                dashSize: 1.5,
                gapSize: 1.0
            });

            activePathMesh = new THREE.Line(geometry, pathMaterial);
            activePathMesh.computeLineDistances(); // Required for animated dash layers
            scene.add(activePathMesh);
        }

        function startRouteSimulation() {
            simulationActive = true;
            
            // Create a physical sphere avatar tracking the active nodes
            if (!simMarker) {
                const markerGeom = new THREE.SphereGeometry(0.8, 16, 16);
                const markerMat = new THREE.MeshStandardMaterial({
                    color: '#10b981',
                    roughness: 0.1,
                    metalness: 0.9,
                    emissive: '#10b981',
                    emissiveIntensity: 0.3
                });
                simMarker = new THREE.Mesh(markerGeom, markerMat);
                scene.add(simMarker);
            }
            simMarker.visible = true;

            const speedFactor = 1500; // Sim walking intervals in milliseconds
            currentSimIndex = 0;

            const runStep = () => {
                if (currentSimIndex >= currentSimPath.length) {
                    stopRouteSimulation();
                    showToast('Destination Reached successfully');
                    return;
                }

                const currentNodeId = currentSimPath[currentSimIndex];
                const nodeData = nodes[currentNodeId];

                // Direct transition animation
                new anime({
                    targets: simMarker.position,
                    x: nodeData.x,
                    y: nodeData.y + 0.8,
                    z: nodeData.z,
                    duration: 900,
                    easing: 'easeInOutQuad'
                });

                document.getElementById('sim-node-counter').innerText = `${currentSimIndex + 1}/${currentSimPath.length}`;
                document.getElementById('telemetry-node').innerText = currentNodeId;

                // Sync elevator portal position with elevator cab animations
                if (currentNodeId.includes('elevator')) {
                    new anime({
                        targets: elevatorCabMesh.position,
                        y: nodeData.y,
                        duration: 800,
                        easing: 'easeInOutCubic'
                    });
                }

                currentSimIndex++;
                simInterval = setTimeout(runStep, speedFactor);
            };

            runStep();
        }

        function stopRouteSimulation() {
            simulationActive = false;
            if (simInterval) {
                clearTimeout(simInterval);
                simInterval = null;
            }
            if (simMarker) {
                simMarker.visible = false;
            }

            const btn = document.getElementById('sim-play-btn');
            const icon = document.getElementById('sim-play-icon');
            const text = document.getElementById('sim-play-text');
            
            btn.className = 'h-8 px-3 rounded-lg bg-brand-cyan text-slate-950 hover:bg-brand-cyan/80 transition-all font-bold text-xs flex items-center gap-1.5';
            icon.className = 'fa-solid fa-play';
            text.innerText = 'Simulate';
        }

        // Trigger dynamic, visual radar structure scan wave pulse
        function triggerAcousticScanPulse() {
            showToast('Initializing active structural scan pulse');

            // Generate temporary glowing wireframe radar spheres in 3D space
            const scanWaveGeom = new THREE.SphereGeometry(1, 32, 32);
            const scanWaveMat = new THREE.MeshBasicMaterial({
                color: '#10b981',
                wireframe: true,
                transparent: true,
                opacity: 0.4
            });
            const scanWave = new THREE.Mesh(scanWaveGeom, scanWaveMat);
            
            // Pulse expands outward from starting location
            const startLoc = nodes[startNodeId];
            scanWave.position.set(startLoc.x, startLoc.y + 0.5, startLoc.z);
            scene.add(scanWave);

            new anime({
                targets: scanWave.scale,
                x: 60,
                y: 60,
                z: 60,
                duration: 2500,
                easing: 'easeOutQuart',
                complete: () => {
                    scene.remove(scanWave);
                    scanWaveGeom.dispose();
                    scanWaveMat.dispose();
                }
            });

            new anime({
                targets: scanWaveMat,
                opacity: 0,
                duration: 2500,
                easing: 'easeOutQuart'
            });
        }

        // Central RequestAnimationFrame loop
        function animate(timestamp) {
            requestAnimationFrame(animate);

            // Update user OrbitControls state
            controls.update();

            // Animate moving dashes along calculated pathlines
            if (activePathMesh && activePathMesh.material) {
                activePathMesh.material.dashOffset -= 0.05;
            }

            // Simple wave floats for pins (adds high tech breathing movement)
            const pulse = Math.sin(Date.now() * 0.003) * 0.15;
            for (const key in waypointVisuals) {
                const pin = waypointVisuals[key];
                if (pin && pin.visible) {
                    pin.position.y = nodes[key].y + 1.5 + pulse;
                    pin.rotation.y += 0.01;
                }
            }

            // Sync rendering frame
            renderer.render(scene, camera);
        }
